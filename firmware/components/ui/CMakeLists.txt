# Product presentation (Architecture Spec §5.12). Owns LVGL.
#
# THIS IS THE ONLY COMPONENT THAT MAY REQUIRE lvgl. LVGL is a private
# requirement, so it is not re-exported: no other component can reach LVGL
# headers through a transitive dependency even by accident, and the public
# include/ui/ui.hpp names no LVGL type.
#
# The internal structure follows §5.12's submodule list:
#
#   src/theme/       ui_theme       tokens, layout, and the binding to LVGL
#   src/fonts/       ui_assets      the generated Inter faces
#   src/formatters   ui_formatters  §17.7's deterministic formatting
#   src/components/  ui_components  §8's shared component library
#   src/scenes/      ui_scene       §9's screen compositions
#   src/controller   ui_controller  scene selection and incremental update
#
# §5.12 also lists ui_projection here. It is implemented in app_model instead, per
# the accepted ADR-0005 boundary: the host test build cannot compile this component
# because it includes LVGL, and §9.4 requires the projection to be host-testable.

# flash-resident glyphs, not an expanded runtime pool or runtime font generator.

endforeach()

# The generated Inter faces (Production UI Specification §4).
#
# Committed C sources, generated by tools/generate-fonts.py from the weights vendored
# at assets/fonts/inter/, with `tnum` frozen into the character map. Compiled in every
# profile: product typography is not optional, and unlike the bring-up scene these are
# what every screen draws.
#
# Listed individually rather than globbed. A glob would silently pick up a face that
# was generated but never referenced, and silently miss one that was deleted — and
# CMake would not reconfigure when the directory changed.
list(APPEND PRD_UI_SRCS
    "src/fonts/prd_font_hero_68.c"

# The Milestone 1 Phase F foundation diagnostic scene, retained as an explicitly
# selected alternative to the product composition (PRD_UI_DIAGNOSTIC_SCENE, default
# n). It is mutually exclusive with the product scene at compile time — see the
# Kconfig help and setup() in src/ui.cpp — so a build contains one scene builder or
# the other and never both.
#
# diagnostic_overlay.cpp joins it on the same terms: it is the owner-paced brightness harness's
# own surface — a fixed grey ramp and two step labels, drawn over the scene above — and it is
# bench instrumentation rather than a product composition. Compiled here and nowhere else, so no
# product image can contain it.
if(CONFIG_PRD_UI_DIAGNOSTIC_SCENE)
    list(APPEND PRD_UI_SRCS "src/validation_scene.cpp")
    list(APPEND PRD_UI_SRCS "src/diagnostic_overlay.cpp")
endif()

# Every C++ TU here includes LVGL and ESP-IDF C headers, which are not -Wundef or
# -Wpedantic clean (undefined macros, anonymous structs, and #include_next -- the
# last cannot be silenced by an in-file diagnostic pragma). Relax only these two
# warnings, per architecture spec §12.3; every other project warning still
# applies. Applied after the policy so it wins.
#
# The generated font sources are excluded from the project policy entirely and given
# their own relaxation below: they are machine output, not project code, so holding
# them to a style policy would mean patching a generator's formatting choices.
set(PRD_UI_POLICY_SRCS ${PRD_UI_SRCS})
list(FILTER PRD_UI_POLICY_SRCS EXCLUDE REGEX "src/fonts/prd_font_.*\\.c$")

set_source_files_properties(
    ${PRD_UI_POLICY_SRCS}
    PROPERTIES
        COMPILE_OPTIONS "-Wno-undef;-Wno-pedantic"
)
